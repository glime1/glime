1) Add this helper immediately BEFORE `async function beginMfaEnrollment(){`

  async function clearPendingAdminFactors(){
    const factors=await sb.auth.mfa.listFactors();
    if(factors.error)throw factors.error;
    const pending=(factors.data?.totp||[]).filter(x=>x.status==='unverified' && x.friendly_name==='GLIME Admin');
    for(const factor of pending){
      const u=await sb.auth.mfa.unenroll({factorId:factor.id});
      if(u.error)throw u.error;
    }
  }

2) Replace the ENTIRE existing `beginMfaEnrollment()` function with:

  async function beginMfaEnrollment(){
    if(mfaEnrollmentBusy)return;
    mfaEnrollmentBusy=true;
    try{
      await clearPendingAdminFactors();

      let r=await sb.auth.mfa.enroll({factorType:'totp',friendlyName:'GLIME Admin'});

      if(r.error && /friendly name|already exists|factor_name_conflict/i.test(r.error.message||'')){
        await clearPendingAdminFactors();
        r=await sb.auth.mfa.enroll({factorType:'totp',friendlyName:'GLIME Admin'});
      }
      if(r.error)throw r.error;

      mfaFactor=r.data;showMfaSetup();
      $('loginQr').innerHTML='<img style="max-width:220px;background:white;padding:10px;border-radius:10px" src="'+r.data.totp.qr_code+'" alt="GLIME Admin TOTP QR">';
      $('loginSecret').textContent='Secret: '+r.data.totp.secret;
      $('setupOtp').focus();
    }catch(e){msg('loginMsg','2FA setup failed: '+e.message);showLogin()}
    finally{mfaEnrollmentBusy=false}
  }
